## 一、云函数公共模块复用方案

### 1.1 背景问题

微信云函数部署时每个函数都是独立环境，无法像普通 Node.js 项目那样直接引用外部文件。导致：
- `getLeaderboard` 和 `exportReport` 中红黑榜计算逻辑重复约 60 行
- 评分常量在多个云函数和前端页面中重复定义

### 1.2 解决方案

采用**构建脚本自动复制**方案：

```
miniprogram/
├── cloudfunctions/
│   ├── shared/              # 公共模块源码
│   │   ├── ranking.js       # 排名计算逻辑
│   │   └── constants.js     # 公共常量定义
│   ├── getLeaderboard/
│   │   ├── shared/          # 构建时自动复制
│   │   └── index.js
│   └── exportReport/
│       ├── shared/          # 构建时自动复制
│       └── index.js
└── scripts/
    └── sync-shared.js       # 同步脚本
```

### 1.3 实现文件

#### scripts/sync-shared.js
```javascript
// 功能：将 cloudfunctions/shared/ 目录复制到各云函数目录
// 使用：node scripts/sync-shared.js
// 目标云函数：getLeaderboard, exportReport, getAnalytics
```

#### cloudfunctions/shared/ranking.js
```javascript
// 导出函数：
// - calculateRanks(records)        // 计算并列排名
// - calculateReverseRanks(records) // 计算倒数排名
// - calculateLeaderboard(records, maxScore) // 计算红黑榜
// - groupByDepartmentAndRoom(records) // 按部门+办公室分组
```

#### cloudfunctions/shared/constants.js
```javascript
// 导出常量：
// - SCORING_ITEMS        // 评分项列表
// - SCORING_MAX_SCORES   // 每项最高分
// - SCORING_LOW_THRESHOLDS // 低分阈值
// - TOTAL_MAX_SCORE      // 总分上限（100）
// - RED_LIST_COUNT       // 红榜数量（3）
// - BLACK_LIST_COUNT     // 黑榜数量（3）
```

```typescript
onLoad() {
```

```typescript
class RequestQueue {

  private processing = 0;
  private maxConcurrent = 5;

  async add<T>(request: () => Promise<T>): Promise<T> {
    return new Promise((resolve, reject) => {
      this.queue.push({ request, resolve, reject });
      this.process();
    });
  }

  private async process() {
    if (this.processing >= this.maxConcurrent || this.queue.length === 0) return;
    this.processing++;
    const item = this.queue.shift()!;
    try {
      const result = await item.request();
      item.resolve(result);
    } catch (err) {
      item.reject(err);
    } finally {
      this.processing--;
      this.process();
    }
  }
}

export function withQueue<T>(request: () => Promise<T>): Promise<T> {
  return requestQueue.add(request);
}
```
